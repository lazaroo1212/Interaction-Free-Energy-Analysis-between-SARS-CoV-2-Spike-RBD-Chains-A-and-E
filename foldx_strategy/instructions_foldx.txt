# --- 1. SETUP & INSTALLATION ---
# Move the FoldX zip file from Downloads to your project folder
mv ~/Downloads/foldx_1Linux64_0.zip ~/Desktop/Bioinformatics3/BIOFISICA/project_biofisica/BioPhysics/Energy_analysis_project/NACCESS/NACCESS/foldx_strategy/

# Unzip the FoldX executable
unzip foldx_1Linux64_0.zip

# Rename the extracted folder/binary to 'foldx' for simplicity
mv foldx_20251231 foldx

# Grant execution permissions to the foldx binary
chmod +x foldx

# --- 2. PRE-PROCESSING (REPAIR) ---
# Minimize the energy of the raw PDB structure to fix Van der Waals clashes (Essential first step)
./foldx --command=RepairPDB --pdb=6m0j.pdb

# --- 3. WILD TYPE INTERACTION ANALYSIS (Step 2a) ---
# Calculate the Interaction Energy (Binding Energy) of the Wild Type (WT) complex
# Chains A and E correspond to ACE2 and Spike RBD respectively
./foldx --command=AnalyseComplex --pdb=6m0j_Repair.pdb --analyseComplexChains=A,E

# --- 4. ALANINE SCANNING (Step 2b) ---
# Run the Python script to format the interface residues list into FoldX format
python3 fix_list.py

# Generate models mutating every interface residue to Alanine
./foldx --command=BuildModel --pdb=6m0j_Repair.pdb --mutant-file=individual_list_corrected.txt

# --- 5. FILE MANAGEMENT ---
# Create a folder to store the many PDB files generated by Alanine Scanning
mkdir ala_scan_pdbs

# Move all generated PDBs to that folder to clean the workspace
mv *_Repair_*.pdb ala_scan_pdbs/

# Restore the original Repaired PDB to the main folder (needed for the next steps)
mv ala_scan_pdbs/6m0j_Repair.pdb .

# Rename/Backup the output files from Alanine Scanning so they are not overwritten later
mv Dif_6m0j_Repair.fxout Dif_AlaScanning.fxout
mv Raw_6m0j_Repair.fxout Raw_AlaScanning.fxout

# --- 6. SARS-CoV-2 VARIANTS ANALYSIS (Step 3) ---
# Create the file containing the specific variants (N501Y, E484K, K417N)
nano variants_list.txt

# Rename the file because FoldX requires the filename to start with 'individual_list'
mv variants_list.txt individual_list_variants.txt

# [ADDED] Generate the 3D models for the specific variants using the corrected list
./foldx --command=BuildModel --pdb=6m0j_Repair.pdb --mutant-file=individual_list_variants.txt

# --- 7. VARIANT INTERACTION ENERGY CALCULATION (Step 4) ---
# Calculate Interaction Energy for Variant 1 (N501Y)
./foldx --command=AnalyseComplex --pdb=6m0j_Repair_1.pdb --analyseComplexChains=A,E

# Calculate Interaction Energy for Variant 2 (E484K)
./foldx --command=AnalyseComplex --pdb=6m0j_Repair_2.pdb --analyseComplexChains=A,E

# Calculate Interaction Energy for Variant 3 (K417N)
./foldx --command=AnalyseComplex --pdb=6m0j_Repair_3.pdb --analyseComplexChains=A,E

# --- 8. FINAL COMPARISON ---
# Extract and display the Interaction Energy values from the summary files for comparison
tail -n 1 Summary_6m0j_Repair*_AC.fxout
